final class FakeAd: NSObject {}
let registry = NapSspFullScreenRegistry<FakeAd>()
var resolved = 0
var rejected = 0
func request() -> NapSspFullScreenRequest {
  NapSspFullScreenRequest(resolve: { _ in resolved += 1 }, reject: { code, _, _ in
    assert(code == "napssp_load_cancelled")
    rejected += 1
  })
}
let first = request()
registry.begin(first, for: "unit")
assert(registry.isLoading("unit"))
registry.cancelLoad("unit")
assert(!registry.isLoading("unit"))
assert(!registry.complete(first, for: "unit"))
first.resolve(nil)
assert(rejected == 1 && resolved == 0)
let second = request()
registry.begin(second, for: "unit")
let staleGeneration = first.generation
assert(!registry.isCurrent(staleGeneration, for: "unit"))
assert(registry.complete(second, for: "unit"))
assert(!registry.complete(second, for: "unit"))
assert(!registry.isLoading("unit"))
let ad = FakeAd()
registry.store(ad, delegate: NSObject(), for: "unit")
second.resolve(nil)
second.resolve(nil)
assert(resolved == 1)
assert(registry.ad(for: "unit") === ad)
assert(registry.cancelLoad("unit") == nil)
assert(registry.ad(for: "unit") === ad)
assert(registry.isCurrent(second.generation, for: "unit"))
assert(registry.isCurrent(second.generation, for: "unit"))
registry.remove("unit")
assert(registry.ad(for: "unit") == nil)
assert(!registry.isCurrent(second.generation, for: "unit"))
assert(rejected == 1)
print("Full-screen registry lifecycle assertions passed")

let superseded = request()
registry.begin(superseded, for: "reload")
registry.remove("reload")
let newest = request()
registry.begin(newest, for: "reload")
assert(!registry.complete(superseded, for: "reload"))
assert(registry.isLoading("reload"))
assert(registry.complete(newest, for: "reload"))
registry.store(ad, delegate: NSObject(), for: "reload")
newest.resolve(nil)
assert(!registry.isCurrent(superseded.generation, for: "reload"))
assert(registry.ad(for: "reload") === ad)
assert(rejected == 2 && resolved == 2)
print("Cancellation preserves ready ads; reload rejects old requests and isolates new generations")
